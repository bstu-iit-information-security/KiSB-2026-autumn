cmake_minimum_required(VERSION 3.16)
project(Lab4Variant12 LANGUAGES CXX)
enable_testing()
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(lab4_core arithmetic.cpp curve.cpp secure.cpp protocols.cpp belt.cpp)
target_compile_features(lab4_core PUBLIC cxx_std_17)
add_executable(lab4_tests tests.cpp)
target_link_libraries(lab4_tests PRIVATE lab4_core)
add_executable(lab4_demo main.cpp)
target_link_libraries(lab4_demo PRIVATE lab4_core)
add_test(NAME lab4_arithmetic COMMAND lab4_tests arithmetic)
add_test(NAME lab4_curve_secure COMMAND lab4_tests curve)
add_test(NAME lab4_protocols COMMAND lab4_tests protocols)
add_test(NAME lab4_demo COMMAND lab4_demo)
add_test(NAME lab4_hex_input COMMAND lab4_demo 00ff12)
add_test(NAME lab4_invalid_hex COMMAND lab4_demo xyz)
set_tests_properties(lab4_invalid_hex PROPERTIES WILL_FAIL TRUE)
if(WIN32)
    target_link_libraries(lab4_core PRIVATE bcrypt)
endif()
if(MSVC)
    target_compile_options(lab4_core PRIVATE /W4 /permissive-)
    target_compile_options(lab4_tests PRIVATE /W4 /permissive-)
    target_compile_options(lab4_demo PRIVATE /W4 /permissive-)
endif()
