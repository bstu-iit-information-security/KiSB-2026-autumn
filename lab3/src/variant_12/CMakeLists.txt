cmake_minimum_required(VERSION 3.16)
project(Lab3Variant12 LANGUAGES CXX)

enable_testing()
add_library(lab3_core belt.cpp modes.cpp)
target_compile_features(lab3_core PUBLIC cxx_std_17)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(lab3_tests tests.cpp)
target_link_libraries(lab3_tests PRIVATE lab3_core)
add_executable(lab3_demo main.cpp)
target_link_libraries(lab3_demo PRIVATE lab3_core)
add_test(NAME lab3_block COMMAND lab3_tests block)
add_test(NAME lab3_modes COMMAND lab3_tests modes)
add_test(NAME lab3_demo COMMAND lab3_demo)
foreach(target lab3_core lab3_tests lab3_demo)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive-)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()
